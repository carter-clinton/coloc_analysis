```python
def _is_symmetric_blocked(m: np.ndarray, atol: float, block: int = 1024) -> bool:
    """Memory-lean symmetry check; equivalent to np.allclose(m, m.T, atol=atol),
    transient bounded to ~block*n_var*4 bytes."""
    n = m.shape[0]
    for i in range(0, n, block):
        a = m[i:i + block, :]
        b = m[:, i:i + block].T
        if not np.allclose(a, b, atol=atol):
            return False
    return True
```

   ```python
               return False
       return True
   ```
