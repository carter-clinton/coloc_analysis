```python
def _existing_region_npz(region_id: str, out_bucket: str | None,
                          out_local_dir: Path | None) -> str | None:
    """Return path/URI to an existing {region_id}.npz, or None.

    Checks in priority order:
      1. {out_bucket}/{region_id}.npz  (production GCS write target)
      2. {out_local_dir or /tmp}/{region_id}.npz  (local-test write target)

    Used by compute_region_ld's W1-G1 idempotency guard. For Path A.3
    (BlockMatrix .bm directory), idempotency would require checking
    {out_bucket}/bm/{region_id}.bm/_SUCCESS — DEFERRED to a follow-up
    quick task; dev-10 has only 2 Path A.3 regions and Wave 2 manual
    re-fire skipping is acceptable for those (re-firing a completed
    .bm write is wasteful but not incorrect since BlockMatrix.write
    accepts overwrite=True).
    """
    if out_bucket is not None:
        candidate = f"{out_bucket}/{region_id}.npz"
        if candidate.startswith("gs://"):
            try:
                import hail as hl
                if hl.hadoop_is_file(candidate):
                    return candidate
            except Exception:
                pass
    # Local fallback
    local = (out_local_dir or Path("/tmp")) / f"{region_id}.npz"
    if local.is_file():
        return str(local)
    return None
```
