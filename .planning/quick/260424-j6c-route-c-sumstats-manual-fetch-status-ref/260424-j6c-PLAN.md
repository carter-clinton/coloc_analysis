    printf 'row\tsource\ttrait\tancestry\tdest\tdir_exists\tfile_count\trel_path\tsize_bytes\tmtime_iso\tsha256\n' >> "$OUT"

    scan_dest() {
      local row="$1" source="$2" trait="$3" ancestry="$4" dest="$5"
      if [[ ! -d "$dest" ]]; then
        printf '%s\t%s\t%s\t%s\t%s\tno\t0\t-\t-\t-\t-\n' \
          "$row" "$source" "$trait" "$ancestry" "$dest" >> "$OUT"

      n=$(find "$dest" -maxdepth 1 -type f ! -name '.*' | wc -l)
      if [[ "$n" -eq 0 ]]; then
        printf '%s\t%s\t%s\t%s\t%s\tyes\t0\t-\t-\t-\t-\n' \
          "$row" "$source" "$trait" "$ancestry" "$dest" >> "$OUT"
        return
      fi
      while IFS= read -r f; do
        local rel sz mt sh
        rel=$(realpath --relative-to=. "$f")
        sz=$(stat -c %s "$f")
        mt=$(stat -c %y "$f" | sed 's/ /T/; s/\..*$//')
        sh=$(sha256sum "$f" | awk '{print $1}')
        printf '%s\t%s\t%s\t%s\t%s\tyes\t%s\t%s\t%s\t%s\t%s\n' \
          "$row" "$source" "$trait" "$ancestry" "$dest" "$n" "$rel" "$sz" "$mt" "$sh" >> "$OUT"
      done < <(find "$dest" -maxdepth 1 -type f ! -name '.*' | sort)
