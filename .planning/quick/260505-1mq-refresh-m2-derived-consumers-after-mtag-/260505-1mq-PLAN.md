### Task 4 — Refresh per-stratum `mtcojo_eligible_targets.tsv` + emit delta sidecars (commit 4: `data`)

Driver: [src/python/select_mtcojo_eligible_targets.py](../../../src/python/select_mtcojo_eligible_targets.py).

```bash
for STRATUM in EUR AFR TRANS; do
    OUTDIR="data/processed/mtcojo/$STRATUM"
    BAK="$OUTDIR/mtcojo_eligible_targets.tsv.preRefresh.bak"
    DELTA="$OUTDIR/mtcojo_eligible_targets.delta.tsv"
    cp "$OUTDIR/mtcojo_eligible_targets.tsv" "$BAK"
    PRE_SHA=$(sha256sum "$OUTDIR/mtcojo_eligible_targets.tsv" | cut -d' ' -f1)
    $PY src/python/select_mtcojo_eligible_targets.py \
        --stratum "$STRATUM" \
        --mtag-filtered "data/processed/mtag/$STRATUM/${STRATUM}_mtag_maxfdr_filtered.txt" \
        --long-matrix "data/processed/ldsc_overlap/rg_matrix_long_M2.tsv" \
        --sidecar "data/processed/mtag/$STRATUM/residcov.trait_order.json" \
        --out "$OUTDIR/mtcojo_eligible_targets.tsv"
    POST_SHA=$(sha256sum "$OUTDIR/mtcojo_eligible_targets.tsv" | cut -d' ' -f1)
    diff <(sort "$BAK") <(sort "$OUTDIR/mtcojo_eligible_targets.tsv") > "$DELTA" || true
    DELTA_LINES=$(wc -l < "$DELTA")
    rm "$BAK"
    echo "$STRATUM pre=$PRE_SHA post=$POST_SHA delta_lines=$DELTA_LINES"
done
```
