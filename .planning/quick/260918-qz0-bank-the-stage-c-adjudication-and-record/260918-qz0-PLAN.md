def md5b(b):
    return hashlib.md5(b).hexdigest()


def apply_edits(text, edits, tag):
    for sid, old, new in edits:
        n = text.count(old)
        if n != 1:
            print("STOP: %s old text occurs %d times in %s (want exactly 1)" % (sid, n, tag))
            return None
        text = text.replace(old, new, 1)
        print("applied %s (%s)" % (sid, tag))
    for sid, old, new in edits:
        if text.count(new) != 1:
            print("STOP: %s new text occurs %d times after all edits (want 1)"
                  % (sid, text.count(new)))
            return None
    return text

def norm(s):
    s = re.sub(r"[*_`]", "", s)
    s = re.sub(r"\s+", " ", s)
    return s.lower()


def normcase(s):
    """like norm() but CASE-PRESERVING. norm() lowercases, so it cannot tell "THE GATES
    WORKING" from "the gates working" - and a quotation that silently re-cases its source is
    still a mis-quotation (W6)."""
    s = re.sub(r"[*_`]", "", s)
    return re.sub(r"\s+", " ", s)


def nc(hay, phrase):
    """count a phrase in ALREADY-NORMALIZED text, normalizing the phrase the same way.
    norm() strips markdown * _ ` - so an underscore inside a symbol name must be stripped on
    BOTH sides, or every symbol check silently counts 0 and passes vacuously."""
    return hay.count(norm(phrase))


def unquote(s):
    return re.sub(r"(?m)^> ?", "", s)


def md5(b):
    return hashlib.md5(b).hexdigest()


def line_of(text, char_idx):
    return text.count("\n", 0, char_idx) + 1


def paragraph_span(lines, anchor):
    spans, start = [], None
    for i, ln in enumerate(lines + [""], start=1):
        if ln.strip() == "":
            if start is not None:
                spans.append((start, i - 1)); start = None
        elif start is None:
            start = i
    return [s for s in spans if norm(unquote("\n".join(lines[s[0] - 1:s[1]]))).find(norm(anchor)) >= 0]


def extract_fence(body):
    ls = body.split("\n")
    f = [i for i, ln in enumerate(ls) if ln == "`" * 4]
    if len(f) != 2:
        return None, len(f)
    return "\n".join(ls[f[0] + 1:f[1]]) + "\n", 2


def census(BL, AL, allowed_ins, R, cid, label):
    """Every changed hunk must be an INSERT after one of `allowed_ins` base lines."""
    sm = difflib.SequenceMatcher(None, BL, AL, autojunk=False)
    parts, changed = [], []
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == "equal":
            continue
        desc = "%s base %d-%d -> after %d-%d" % (tag, i1 + 1, i2, j1 + 1, j2)
        parts.append(desc)
        R.append((cid, tag == "insert" and i1 in allowed_ins, desc))
        if tag in ("replace", "insert"):
            changed.append((j1 + 1, "\n".join(AL[j1:j2])))
    print("  census %s: %s" % (label, "; ".join(parts)))
    R.append((cid + "-nonempty", bool(parts), "%s: %d changed hunk(s)" % (label, len(parts))))
    return changed

def classify_numbers(text, allowed, label, R, cid):
    t = text.lower()
    for pat in STRUCT_RE:
        t = re.sub(pat, " ", t)
    bad = []
    for m in NUM_RE.finditer(t):
        tok = m.group(0).replace(",", "")
        if not allowed.get(tok, False):
            bad.append(tok)
    R.append((cid, not bad, "%s: unreconciled numeric tokens %s" % (label, sorted(set(bad))) if bad
              else "%s: every numeric token reconciled" % label))


def explain_enum(blocks, label, R, cid):
    print("  enumeration %s (explain family in %s):" % (cid, label))
    n = 0
    for first, blk in blocks:
        bn = norm(unquote(blk))
        for m in re.finditer(EXPLAIN_RE, bn):
            n += 1
            pre = bn[max(0, m.start() - 70):m.start()]
            print("    block@%-5d ...%s[%s]  NOT NEGATED" % (first, pre[-45:], m.group(0)))
            R.append((cid, False, "block@%d %r -> NOT NEGATED" % (first, pre[-40:] + m.group(0))))
    print("    -> %d occurrence(s) in %s (want 0)" % (n, label))
    R.append((cid, n == 0, "%s carries %d explain-family occurrence(s) (want 0)" % (label, n)))
    return n

def mutate_cites(op):
    c = copy.deepcopy(CIT0)
    if op == "digest":
        c["digest"] = "0" * 32
    elif op == "inject-moving":
        c["rows"].append(["ap", ".planning/STATE.md", 999, "explicit", 660, 665])
    elif op == "drop-rows":
        c["rows"] = []
    elif op == "bare-digest":
        c["bare_digest"] = "0" * 32
    elif op == "bare-inject":
        c["bare"].append([".planning/STATE.md", 999, 650, "injected bare reference"])
    elif op == "unt-digest":
        c["unt_digest"] = "0" * 32
    elif op == "unt-inject-live-moved":
        # a LIVE stale locator: a citation into the moving range whose own line does NOT
        # carry the post-shift range, i.e. it points at the old place and nothing else.
        c["unt"].append(["ap", ".planning/STATE.md", 999, "explicit", 660, 665,
                         "see AGENT-PROMPT.md:660-665 for the gate reading"])
    elif op == "unt-drop-untracked":
        # leave pass C seeing only files pass A already sees -> the pass is vacuous
        seen_a = {r[1] for r in c["rows"]}
        c["unt"] = [r for r in c["unt"] if r[1] in seen_a]
    elif op == "inject-oor":
        c["rows"].append(["def", ".planning/STATE.md", 999, "explicit", 9000, 9001])
    else:
        raise SystemExit("unknown cites op " + op)
    return c


def run_with(tag, files, cites):
    d = os.path.join(work, tag)
    os.makedirs(d, exist_ok=True)
    ps = {}
    for k, body in files.items():
        p = os.path.join(d, FN[k])
        with open(p, "w", encoding="utf-8") as fh:
            fh.write(body)
        ps[k] = p
    cp = os.path.join(d, "cites.json")
    json.dump(cites, open(cp, "w"))
    with contextlib.redirect_stdout(io.StringIO()):
        R = G.run(ps["bosf"], ps["osf"], ps["bdec"], ps["dec"], ps["bap"], ps["ap"],
                  ps["bdef"], ps["def"], ps["bank"], ps["adj"], ps["v2"], ps["halt"],
                  ps["tests"], cp)
    return R


def main():
    bad = 0
    R0 = run_with("positive", dict(F0), CIT0)
    red0 = sorted({cid for cid, ok, _ in R0 if not ok})
    all_ids = sorted({cid for cid, _, _ in R0})
    print("POSITIVE CONTROL (unmutated): %d checks, %d ids, FAIL ids = %s -> %s"
          % (len(R0), len(all_ids), red0, "GREEN" if not red0 else "NOT GREEN"))
    bad += bool(red0)
    covered = set()
    for cid, which, old, new, expect in CONTROLS:
        files = dict(F0)
        cites = CIT0
        if which == "cites":
            cites = mutate_cites(old)
        elif which.startswith("identity-"):
            k = which.split("-", 1)[1]
            files[k] = files["b" + k]
        else:
            src = files[which]
            n = src.count(old)
            if n != 1:
                print("%s  INVALID CONTROL: old text occurs %d times in %s" % (cid, n, which))
                bad += 1
                continue
            files[which] = src.replace(old, new, 1)
        R = run_with(cid, files, cites)
        red = sorted({c for c, ok, _ in R if not ok})
        covered |= set(red)
        missing = [e for e in expect if e not in red]
        print("%s  %-14s expect RED %s -> observed RED %s  %s"
              % (cid, which, expect, red, "OK" if not missing else "MISSING %s" % missing))
        bad += bool(missing)
    uncovered = [i for i in all_ids if i not in covered and i not in UNFALSIFIABLE]
    print("\nCOVERAGE: %d/%d check ids observed RED by at least one control"
          % (len(all_ids) - len(uncovered), len(all_ids)))
    if uncovered:
        print("  UNCOVERED (no control, no UNFALSIFIABLE label): %s" % uncovered)
        bad += 1
    for k, v in UNFALSIFIABLE.items():
        print("  UNFALSIFIABLE %-24s %s" % (k, v))
    print("\nNEGCTL RESULT: %s" % ("ALL CONTROLS OBSERVED RED, ALL IDS COVERED" if not bad
                                   else "%d PROBLEM(S)" % bad))
    return 0 if not bad else 1


if __name__ == "__main__":
    sys.exit(main())
