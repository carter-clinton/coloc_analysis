      ```r
      apply_majority_rule <- function(row, alpha) {
        methods <- c("ivw", "egger", "median", "presso", "raps")
        pvals <- sapply(methods, function(m) row[[paste0(m, "_pval")]])
        betas <- sapply(methods, function(m) row[[paste0(m, "_beta")]])
        # Exclude NA methods from count
        valid <- !is.na(pvals)
        if (sum(valid) == 0) return(list(verdict="null", n_sig=0, n_concordant=0))
        significant <- pvals[valid] < alpha
        if (sum(significant) == 0) return(list(verdict="null", n_sig=0, n_concordant=0))
        directions <- sign(betas[valid][significant])
        dir_table <- table(directions)
        dominant_dir <- as.integer(names(dir_table)[which.max(dir_table)])
        n_concordant <- max(dir_table)
        verdict <- if (n_concordant >= 3) "strong"
                   else if (n_concordant >= 2) "moderate"
                   else "suggestive"
        list(verdict=verdict, n_sig=sum(significant), n_concordant=n_concordant,
             dominant_direction=dominant_dir)
      }
      ```
      Use bonferroni_bidirectional (0.0025) for bidirectional tests, bonferroni_mvmr (0.017) for MVMR.
