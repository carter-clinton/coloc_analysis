    # --- Phase 9 additions (2026-04-13) ---
    PALINDROMIC_PAIRS = {("A","T"), ("T","A"), ("C","G"), ("G","C")}

    def is_palindromic(ea: str, oa: str) -> bool:
        return (ea.upper(), oa.upper()) in PALINDROMIC_PAIRS

    def filter_palindromic_ambiguous(
        df: pd.DataFrame, ea_col: str = "EA", oa_col: str = "OA",
        eaf_col: str = "EAF", maf_band: tuple = (0.48, 0.52)
    ) -> pd.DataFrame:
        """Drop palindromic SNPs with MAF in ambiguity band (RESEARCH pitfall #2).
        Keep all others but tag `palindromic_flag`."""
        pal_mask = df.apply(lambda r: is_palindromic(r[ea_col], r[oa_col]), axis=1)
        maf = df[eaf_col].where(df[eaf_col] < 0.5, 1 - df[eaf_col])
        ambig = pal_mask & maf.between(maf_band[0], maf_band[1])
        df = df.loc[~ambig].copy()
        df["palindromic_flag"] = pal_mask[~ambig].values
        return df

    def liftover_to_grch37(df: pd.DataFrame, chain_file: str) -> tuple[pd.DataFrame, dict]:
        """Apply GRCh38→37 liftover. Returns (lifted_df, qc_dict) where qc_dict
        has n_input, n_lifted, n_dropped, drop_rate. Fails hard if drop_rate > 0.05."""
        from src.python.liftover import liftover_coordinates
        n_in = len(df)
        new_pos = df.apply(lambda r: liftover_coordinates(chain_file, str(r["CHR"]), int(r["BP"])), axis=1)
        mask = new_pos.notna()
        df_lifted = df.loc[mask].copy()
        lifted_coords = new_pos[mask].apply(lambda t: pd.Series({"CHR_new": t[0], "BP_new": t[1]}))
        df_lifted["CHR"] = lifted_coords["CHR_new"].astype(str).str.replace("chr", "")
        df_lifted["BP"] = lifted_coords["BP_new"].astype(int)
        qc = {
            "n_input": n_in, "n_lifted": int(mask.sum()),
            "n_dropped": int((~mask).sum()),
            "drop_rate": float((~mask).mean())
        }
        if qc["drop_rate"] > 0.05:
            raise RuntimeError(f"Liftover drop rate {qc['drop_rate']:.2%} exceeds 5% threshold (RESEARCH pitfall #1)")
        return df_lifted, qc
    ```

    """FinnGen R12 TSV → canonical harmonized sumstats."""
    from pathlib import Path
    import pandas as pd
    import yaml
    from src.python.sumstats_utils import (
        filter_palindromic_ambiguous, liftover_to_grch37,
    )

    CANONICAL_COLS = ["CHR","BP","SNP","EA","OA","BETA","SE","P","EAF","N"]

    def harmonize_finngen_sumstats(
        input_gz: Path, output_gz: Path, chain_file: Path,
        trait: str, case_n: int, ctrl_n: int = None, qc_out: Path = None,
    ) -> dict:
        """Read FinnGen R12 tsv.gz, rename to canonical, liftover, drop palindromic.
        Raises on liftover drop rate > 5%.
        Returns QC dict: {n_input, n_lifted, n_after_palindromic, drop_rate}.
        """
        # FinnGen schema (per config/replication_cohorts.yaml):
        #   #chrom pos ref alt rsids nearest_genes pval mlogp beta sebeta af_alt af_alt_cases af_alt_controls
        df = pd.read_csv(input_gz, sep="\t", compression="gzip", low_memory=False)
        df = df.rename(columns={
            "#chrom": "CHR", "pos": "BP", "rsids": "SNP",
            "ref": "OA", "alt": "EA",
            "beta": "BETA", "sebeta": "SE", "pval": "P", "af_alt": "EAF",
        })
        # N: FinnGen publishes case_n + ctrl_n per endpoint; passed in by caller
        n_total = case_n + (ctrl_n or case_n)  # caller must pass ctrl_n for binary
        df["N"] = n_total
        df["N_CASES"] = case_n
        if ctrl_n is not None:
            df["N_CTRLS"] = ctrl_n
        df = df[CANONICAL_COLS + [c for c in ["N_CASES","N_CTRLS"] if c in df.columns]]
        # Liftover GRCh38 → GRCh37
        df_lifted, liftover_qc = liftover_to_grch37(df, str(chain_file))
        # Palindromic exclusion (RESEARCH pitfall #2)
        n_before_pal = len(df_lifted)
        df_out = filter_palindromic_ambiguous(df_lifted)
        qc = {
            **liftover_qc,
            "n_after_palindromic": len(df_out),
            "n_palindromic_dropped": n_before_pal - len(df_out),
            "trait": trait, "cohort": "finngen_r12",
        }
        df_out.to_csv(output_gz, sep="\t", index=False, compression="gzip")
        if qc_out:
            import json
            Path(qc_out).write_text(json.dumps(qc, indent=2))
        return qc

    if __name__ == "__main__":
        import argparse
        ap = argparse.ArgumentParser()
        ap.add_argument("--input", required=True)
        ap.add_argument("--output", required=True)
        ap.add_argument("--chain-file", required=True)
        ap.add_argument("--trait", required=True)
        ap.add_argument("--case-n", type=int, required=True)
        ap.add_argument("--ctrl-n", type=int, default=None)
        ap.add_argument("--qc-out", default=None)
        args = ap.parse_args()
        harmonize_finngen_sumstats(
            Path(args.input), Path(args.output), Path(args.chain_file),
            args.trait, args.case_n, args.ctrl_n, args.qc_out,
        )

    """GBMI TSV → per-ancestry canonical harmonized sumstats.
    GBMI publishes one file per trait with all-ancestry meta columns + per-ancestry columns."""
    from pathlib import Path
    import pandas as pd
    from src.python.sumstats_utils import filter_palindromic_ambiguous

    CANONICAL_COLS = ["CHR","BP","SNP","EA","OA","BETA","SE","P","EAF","N"]
    ANCESTRIES = ["eur", "afr", "eas", "amr", "sas"]

    def harmonize_gbmi_sumstats(
        input_gz: Path, output_prefix: Path, trait: str, ancestry: str = "eur",
    ) -> dict:
        """Extract {ancestry}-specific columns + meta columns from a GBMI trait file.
        Emits {output_prefix}_{ancestry}.tsv.gz per ancestry.
        Note: GBMI is already GRCh37 — no liftover needed (RESEARCH §4)."""
        df = pd.read_csv(input_gz, sep="\t", compression="gzip", low_memory=False)
        prefix = f"{ancestry}_meta" if ancestry != "eur" else "all_meta"  # verify at execute time

        # GBMI schema (per config/replication_cohorts.yaml for 'all_meta' = all-ancestry):
        col_map = {
            "CHR": "CHR", "POS": "BP", "rsid": "SNP", "REF": "OA", "ALT": "EA",
            f"{prefix}_beta": "BETA", f"{prefix}_sebeta": "SE", f"{prefix}_pval": "P",
            f"{prefix}_AF": "EAF", f"{prefix}_sample_N": "N",
        }
        missing = [src for src in col_map if src not in df.columns]
        if missing:
            raise ValueError(f"GBMI {ancestry} stratum missing columns: {missing}")
        df = df[list(col_map)].rename(columns=col_map)
        df = df[CANONICAL_COLS]
        df = filter_palindromic_ambiguous(df)

        out = output_prefix.parent / f"{output_prefix.name}_{ancestry}.tsv.gz"
        df.to_csv(out, sep="\t", index=False, compression="gzip")
        return {
            "cohort": f"gbmi_{ancestry}", "trait": trait, "ancestry": ancestry,
            "n_rows": len(df), "output": str(out),
        }
    ```

    """MVP dbGaP phs001672 REGENIE/SAIGE output → canonical.
    MVP is mostly GRCh37 already (verify per-file in inventory)."""
    from pathlib import Path
    import pandas as pd
    import numpy as np
    from src.python.sumstats_utils import filter_palindromic_ambiguous, liftover_to_grch37

    CANONICAL_COLS = ["CHR","BP","SNP","EA","OA","BETA","SE","P","EAF","N"]

    def log10p_to_pval(log10p: pd.Series, clip: tuple = (0, 300)) -> pd.Series:
        """Phase 2 convention: clip LOG10P to [0, 300] (avoid underflow at P=1e-300)."""
        clipped = log10p.clip(lower=clip[0], upper=clip[1])
        return np.power(10.0, -clipped)

    def harmonize_mvp_sumstats(
        input_gz: Path, output_gz: Path, pha_id: str, trait: str, ancestry: str,
        genome_build: str = "GRCh37", chain_file: Path = None, qc_out: Path = None,
    ) -> dict:
        df = pd.read_csv(input_gz, sep="\t", compression="gzip", low_memory=False)
        df = df.rename(columns={
            "CHROM": "CHR", "POS": "BP", "ID": "SNP",
            "REF": "OA", "ALT": "EA", "A1_FREQ": "EAF",
            "BETA": "BETA", "SE": "SE", "N": "N",
        })
        if "LOG10P" in df.columns:
            df["P"] = log10p_to_pval(df["LOG10P"])
        elif "PVAL" in df.columns:
            df["P"] = df["PVAL"]
        else:
            raise ValueError(f"MVP {pha_id}: no LOG10P or PVAL column found")
        df = df[CANONICAL_COLS]
        qc = {"n_input": len(df), "pha_id": pha_id, "trait": trait, "ancestry": ancestry}
        if genome_build == "GRCh38":
            assert chain_file is not None, "MVP GRCh38 file requires chain_file"
            df, liftover_qc = liftover_to_grch37(df, str(chain_file))
            qc.update(liftover_qc)
        df = filter_palindromic_ambiguous(df)
        qc["n_output"] = len(df)
        df.to_csv(output_gz, sep="\t", index=False, compression="gzip")
        if qc_out:
            import json; Path(qc_out).write_text(json.dumps(qc, indent=2))
        return qc
    ```

                          if n.lower().endswith((".tsv", ".txt"))
                          and "readme" not in n.lower()]
            if not candidates:
                raise ValueError(f"{zip_path}: no .tsv/.txt file found in zip")
            zf.extractall(out_dir)
            return out_dir / candidates[0]

    def harmonize_bbj_sumstats(
        input_file: Path, output_gz: Path, chain_file: Path,
        trait: str, trait_code: str, case_n: int = None, ctrl_n: int = None,
        qc_out: Path = None,
    ) -> dict:
        # BBJ may be tsv or txt; auto-detect separator
        df = pd.read_csv(input_file, sep=None, engine="python", low_memory=False)
        df = df.rename(columns={
            "CHR": "CHR", "POS": "BP", "SNPID": "SNP",
            "Allele1": "OA", "Allele2": "EA", "AF": "EAF",
            "Beta": "BETA", "SE": "SE", "p.value": "P", "N": "N",
        })
        df = df[CANONICAL_COLS]
        df, liftover_qc = liftover_to_grch37(df, str(chain_file))
        df = filter_palindromic_ambiguous(df)
        qc = {"cohort": "bbj_hum0197_v3", "trait": trait, "trait_code": trait_code, **liftover_qc, "n_output": len(df)}
        df.to_csv(output_gz, sep="\t", index=False, compression="gzip")
        if qc_out:
            import json; Path(qc_out).write_text(json.dumps(qc, indent=2))
        return qc

    """Canonical-schema validator + liftover QC gate."""
    from pathlib import Path
    import pandas as pd, json, sys

    CANONICAL = ["CHR","BP","SNP","EA","OA","BETA","SE","P","EAF","N"]

    def validate_schema(tsv_gz: Path) -> dict:
        df = pd.read_csv(tsv_gz, sep="\t", compression="gzip", nrows=100)
        missing = [c for c in CANONICAL if c not in df.columns]
        return {"valid": not missing, "missing": missing, "columns": list(df.columns)}

    def check_liftover_qc(qc_json: Path, max_drop: float = 0.05) -> bool:
        qc = json.loads(qc_json.read_text())
        if qc.get("drop_rate", 0) > max_drop:
            return False
        return True

    if __name__ == "__main__":
        import argparse
        ap = argparse.ArgumentParser()
        ap.add_argument("--tsv", required=True)
        ap.add_argument("--qc", default=None)
        ap.add_argument("--max-drop", type=float, default=0.05)
        args = ap.parse_args()
        res = validate_schema(Path(args.tsv))
        if not res["valid"]:
            print(f"FAIL schema: missing {res['missing']}"); sys.exit(1)
        if args.qc and not check_liftover_qc(Path(args.qc), args.max_drop):
            print(f"FAIL liftover QC (> {args.max_drop})"); sys.exit(1)
        print("OK"); sys.exit(0)
    ```
