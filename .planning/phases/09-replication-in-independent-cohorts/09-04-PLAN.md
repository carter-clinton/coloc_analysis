    ```r
    #!/usr/bin/env Rscript
    # Run coloc.susie(disc_fit, rep_fit) and emit PP.H4 sweep per D-03b.
    # Output JSON schema matches Phase 1 run_coloc_susie.R for downstream consumer compatibility.

    suppressPackageStartupMessages({
      library(coloc); library(jsonlite)
    })

    run_replication_coloc <- function(
      disc_fit_path, rep_fit_path, signal_id, cohort,
      pph4_thresholds = c(0.5, 0.7, 0.8, 0.9),
      output_json
    ) {
      disc_fit <- readRDS(disc_fit_path)
      rep_fit  <- readRDS(rep_fit_path)

      res <- tryCatch(
        coloc::coloc.susie(disc_fit, rep_fit),
        error = function(e) { list(summary = NULL, error = conditionMessage(e)) }
      )

      if (is.null(res$summary)) {
        out <- list(
          signal_id = signal_id, cohort = cohort,
          coloc_succeeded = FALSE, error = res$error,
          pph4_best = NA, susie_pairs = list()
        )
      } else {
        best_idx <- which.max(res$summary$PP.H4.abf)
        pph4_best <- res$summary$PP.H4.abf[best_idx]
        sweep <- setNames(
          as.list(pph4_best >= pph4_thresholds),
          sprintf("replicated_pph4_%.1f", pph4_thresholds)
        )
        out <- c(
          list(
            signal_id = signal_id, cohort = cohort,
            coloc_succeeded = TRUE,
            pph4_best = pph4_best,
            best_hit1 = res$summary$hit1[best_idx],
            best_hit2 = res$summary$hit2[best_idx],
            nsnps = res$summary$nsnps[best_idx],
            susie_pairs = res$summary  # full pairwise preserved
          ),
          sweep
        )
      }
      jsonlite::write_json(out, output_json, auto_unbox = TRUE, pretty = TRUE)
      invisible(out)
    }

    # M-2 revision: define %||% BEFORE the CLI block that uses it.
    `%||%` <- function(a,b) if (is.null(a) || is.na(a) || identical(a, "")) b else a

    if (!interactive() && sys.nframe() == 0L) {
      args <- commandArgs(trailingOnly = TRUE)
      nm <- setNames(sapply(strsplit(args, "="), `[`, 2), sapply(strsplit(args, "="), `[`, 1))
      run_replication_coloc(
        disc_fit_path = nm[["disc"]],
        rep_fit_path  = nm[["rep"]],
        signal_id     = nm[["signal_id"]],
        cohort        = nm[["cohort"]],
        pph4_thresholds = as.numeric(strsplit(nm[["thresholds"]] %||% "0.5,0.7,0.8,0.9", ",")[[1]]),
        output_json   = nm[["out"]]
      )
    }
    ```

    ```python
    """Per-cohort Bonferroni effect-size test + same-direction check + post-hoc power.
    D-03a + RESEARCH pitfall #4 + #5."""
    from pathlib import Path
    import pandas as pd, numpy as np
    from scipy.stats import norm

    def compute_bonferroni(n_signals_in_cohort: int, alpha: float = 0.05) -> float:
        """RESEARCH pitfall #4: denominator = N_signals_tested_in_THIS_cohort."""
        if n_signals_in_cohort < 1:
            raise ValueError("n_signals must be >= 1")
        return alpha / n_signals_in_cohort

    def check_same_direction(beta_disc: float, beta_rep: float) -> bool:
        """D-03a: same-direction β required for effect-size criterion."""
        if pd.isna(beta_disc) or pd.isna(beta_rep):
            return False
        return np.sign(beta_disc) == np.sign(beta_rep)

    def posthoc_power(beta_fiqt: float, se_rep: float, alpha: float) -> float:
        """Two-sided post-hoc power at α for detecting β_fiqt using replication SE."""
        if pd.isna(beta_fiqt) or pd.isna(se_rep) or se_rep <= 0:
            return float("nan")
        z_alpha = norm.ppf(1 - alpha / 2)
        z_beta = abs(beta_fiqt) / se_rep - z_alpha
        return float(norm.cdf(z_beta))

    def compute_joint_criterion(row: pd.Series, primary_threshold: float = 0.8) -> bool:
        """D-03a: joint = Bonferroni effect AND coloc pph4 >= primary_threshold."""
        key_pph4 = f"replicated_pph4_{primary_threshold}"
        return bool(row.get("replicated_bonferroni", False) and row.get(key_pph4, False))

    def process_cohort(
        effect_df: pd.DataFrame, fiqt_df: pd.DataFrame, coloc_df: pd.DataFrame,
        cohort: str, primary_threshold: float = 0.8,
    ) -> pd.DataFrame:
        """Merge FIQT discovery + per-cohort replication effect + coloc sweep;
        compute Bonferroni, same-direction, post-hoc power, joint."""
        df = effect_df.merge(fiqt_df, on="signal_id").merge(coloc_df, on=["signal_id","cohort"])
        n_in_cohort = df["signal_id"].nunique()
        alpha_bonf = compute_bonferroni(n_in_cohort)
        df["bonf_threshold"] = alpha_bonf
        df["same_direction"] = df.apply(lambda r: check_same_direction(r["beta_discovery_FIQT"], r["beta_replication"]), axis=1)
        df["replicated_bonferroni"] = (df["p_replication"] < alpha_bonf) & df["same_direction"]
        df["power_posthoc"] = df.apply(lambda r: posthoc_power(r["beta_discovery_FIQT"], r["se_replication"], alpha_bonf), axis=1)
        df[f"replicated_joint_{primary_threshold}"] = df.apply(lambda r: compute_joint_criterion(r, primary_threshold), axis=1)
        return df

    if __name__ == "__main__":
        import argparse
        ap = argparse.ArgumentParser()
        ap.add_argument("--effect", required=True)
        ap.add_argument("--fiqt", required=True)
        ap.add_argument("--coloc", required=True)
        ap.add_argument("--cohort", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        df = process_cohort(
            pd.read_csv(a.effect, sep="\t"),
            pd.read_csv(a.fiqt, sep="\t"),
            pd.read_csv(a.coloc, sep="\t"),
            a.cohort,
        )
        df.to_csv(a.out, sep="\t", index=False)
    ```

    Step 1b — `src/python/collect_replication_effect_sizes.py` (I-5 revision: producer for effect_size_raw/{cohort}.tsv):
    ```python
    """Per-cohort effect-size extractor. Reads manifest + harmonized sumstats for a given
    cohort; emits one row per manifest signal_id with (beta_replication, se_replication,
    p_replication, eaf_replication, N) pulled from the harmonized sumstats file at the
    manifest's lead SNP or credible-set-spanning region lookup."""
    from pathlib import Path
    import pandas as pd, argparse

    CANONICAL_COLS = ["CHR","BP","SNP","EA","OA","BETA","SE","P","EAF","N"]

    def extract_per_signal(manifest_df: pd.DataFrame, cohort: str, sumstats_resolver) -> pd.DataFrame:
        """For each manifest row where cohort == `cohort`, open that row's
        replication_sumstats_path, locate the lead SNP or region, and extract
        (beta, se, p, eaf, N). Returns wide DataFrame with columns:
        signal_id, cohort, beta_replication, se_replication, p_replication, eaf_replication, N."""
        rows = []
        cohort_rows = manifest_df[manifest_df["cohort"] == cohort]
        for _, r in cohort_rows.iterrows():
            path = Path(r["replication_sumstats_path"])
            if not path.exists():
                rows.append({"signal_id": r["signal_id"], "cohort": cohort,
                             "beta_replication": None, "se_replication": None,
                             "p_replication": None, "eaf_replication": None, "N": None})
                continue
            ss = pd.read_csv(path, sep="\t", compression="gzip")
            # Match on lead SNP (rsid) first, then chr:bp fallback
            lead = r.get("lead_snp") or r.get("SNP")
            match = ss[ss["SNP"] == lead]
            if match.empty and "lead_chr" in r and "lead_bp" in r:
                match = ss[(ss["CHR"].astype(str) == str(r["lead_chr"])) & (ss["BP"] == int(r["lead_bp"]))]
            if match.empty:
                rows.append({"signal_id": r["signal_id"], "cohort": cohort,
                             "beta_replication": None, "se_replication": None,
                             "p_replication": None, "eaf_replication": None, "N": None})
            else:
                m = match.iloc[0]
                rows.append({
                    "signal_id": r["signal_id"], "cohort": cohort,
                    "beta_replication": float(m["BETA"]),
                    "se_replication": float(m["SE"]),
                    "p_replication": float(m["P"]),
                    "eaf_replication": float(m["EAF"]),
                    "N": int(m["N"]) if pd.notna(m["N"]) else None,
                })
        return pd.DataFrame(rows)

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()
        ap.add_argument("--manifest", required=True)
        ap.add_argument("--cohort", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        manifest = pd.read_csv(a.manifest, sep="\t")
        df = extract_per_signal(manifest, a.cohort, sumstats_resolver=None)
        df.to_csv(a.out, sep="\t", index=False)
    ```

    Step 2 — `src/snakemake/scripts/aggregate_replication_meta.R`:
    ```r
    #!/usr/bin/env Rscript
    # IVW fixed-effect meta-analysis across ancestry-matched cohorts per signal.
    # D-06b: metafor::rma.uni(method='FE')

    suppressPackageStartupMessages({
      library(metafor); library(data.table)
    })

    ivw_meta_per_signal <- function(per_cohort_df, min_cohorts = 2) {
      # Input rows: cohort, beta_replication, se_replication
      df <- per_cohort_df[!is.na(beta_replication) & !is.na(se_replication) & se_replication > 0]
      if (nrow(df) < min_cohorts) return(NULL)
      res <- tryCatch(
        metafor::rma.uni(yi = df$beta_replication, sei = df$se_replication, method = "FE"),
        error = function(e) NULL
      )
      if (is.null(res)) return(NULL)
      data.table(
        signal_id = df$signal_id[1],
        meta_ancestry = df$cohort_ancestry[1],
        beta_meta = as.numeric(res$beta),
        se_meta = as.numeric(res$se),
        p_meta = as.numeric(res$pval),
        meta_n_cohorts_contributing = nrow(df)
      )
    }

    aggregate_ivw <- function(per_cohort_tsv, output_tsv) {
      df <- fread(per_cohort_tsv)
      res <- df[, ivw_meta_per_signal(.SD), by = signal_id, .SDcols = c("cohort","cohort_ancestry","beta_replication","se_replication")]
      # Flag meta Bonferroni at cohort-count-based denominator
      n_sig <- uniqueN(res$signal_id)
      res[, meta_replicated_bonferroni := p_meta < (0.05 / n_sig)]
      fwrite(res, output_tsv, sep = "\t")
      invisible(res)
    }

    if (!interactive() && sys.nframe() == 0L) {
      args <- commandArgs(trailingOnly = TRUE)
      aggregate_ivw(args[1], args[2])
    }
    ```
