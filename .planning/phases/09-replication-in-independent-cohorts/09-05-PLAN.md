    """Convert canonical harmonized sumstats to COJO .ma format."""
    import pandas as pd, sys, argparse
    from pathlib import Path

    def canonical_to_ma(canonical_tsv: Path, locus_region: str, output_ma: Path) -> int:
        """locus_region format: 'chr:start-end' (GRCh37)."""
        chrom, range_ = locus_region.replace("chr", "").split(":")
        start, end = [int(x) for x in range_.split("-")]
        df = pd.read_csv(canonical_tsv, sep="\t", compression="gzip" if str(canonical_tsv).endswith(".gz") else None)
        df = df[(df["CHR"].astype(str) == chrom) & (df["BP"] >= start) & (df["BP"] <= end)]
        ma = pd.DataFrame({
            "SNP": df["SNP"], "A1": df["EA"], "A2": df["OA"],
            "freq": df["EAF"], "b": df["BETA"], "se": df["SE"],
            "p": df["P"], "N": df["N"],
        })
        ma.to_csv(output_ma, sep=" ", index=False)
        return len(ma)

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()
        ap.add_argument("--sumstats", required=True)
        ap.add_argument("--region", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        n = canonical_to_ma(Path(a.sumstats), a.region, Path(a.out))
        print(f"wrote {n} SNPs to {a.out}")
    ```

    # COJO --cojo-slct wrapper per D-04c + RESEARCH §6 Option D.
    # Usage: run_cojo.sh <ma_file> <plink_prefix> <locus_snp_list> <out_prefix>
    set -euo pipefail

    PLINK_PREFIX="$2"   # e.g., data/processed/ld_reference/1kg_eur/chr10
    LOCUS_SNPS="$3"     # file with one SNP rsid per line
    OUT_PREFIX="$4"

    # Sample-size sanity check (RESEARCH §6 Option D caveat)
    N_SAMPLES=$(wc -l < "${PLINK_PREFIX}.fam")
    if [ "$N_SAMPLES" -lt 4000 ]; then
      echo "WARN: LD reference $PLINK_PREFIX has N=$N_SAMPLES (< 4000 GCTA threshold). COJO results are tier-2 sensitivity per RESEARCH §6 Option D." >&2
    fi

    gcta \
      --bfile "$PLINK_PREFIX" \
      --cojo-file "$MA" \
      --cojo-slct \
      --cojo-p 5e-8 \
      --cojo-wind 10000 \
      --extract "$LOCUS_SNPS" \
      --out "$OUT_PREFIX"

    """Parse all .jma.cojo outputs into cojo_sensitivity.tsv (D-04c supplementary)."""
    from pathlib import Path
    import pandas as pd, glob, re, argparse

    def parse_cojo_jma(jma_path: Path, signal_id: str, cohort: str) -> dict:
        """Parse a single COJO .jma.cojo output."""
        try:
            df = pd.read_csv(jma_path, sep="\t")
        except Exception as e:
            return {"signal_id": signal_id, "cohort": cohort, "cojo_n_independent_signals": 0,
                    "cojo_top_snp": None, "cojo_joint_beta": None, "cojo_joint_p": None,
                    "note": f"parse failed: {e}"}
        n_indep = len(df)
        if n_indep == 0:
            return {"signal_id": signal_id, "cohort": cohort, "cojo_n_independent_signals": 0,
                    "cojo_top_snp": None, "cojo_joint_beta": None, "cojo_joint_p": None}
        top = df.iloc[df["pJ"].idxmin()] if "pJ" in df.columns else df.iloc[0]
        return {
            "signal_id": signal_id, "cohort": cohort,
            "cojo_n_independent_signals": n_indep,
            "cojo_top_snp": top.get("SNP"),
            "cojo_joint_beta": top.get("bJ"),
            "cojo_joint_p": top.get("pJ"),
            "secondary_signal_notes": f"top-SNP LD_r to lead = {top.get('LD_r', 'NA')}",
        }

    def build_cojo_table(cojo_dir: Path, manifest_tsv: Path, output_tsv: Path):
        manifest = pd.read_csv(manifest_tsv, sep="\t")
        rows = []
        for _, m in manifest.iterrows():
            jma = cojo_dir / f"{m['cohort']}_{m['discovery_trait']}_{m['region'].replace(':','_').replace('-','_')}.jma.cojo"
            if not jma.exists():
                continue
            rows.append(parse_cojo_jma(jma, m["signal_id"], m["cohort"]))
        df = pd.DataFrame(rows)
        df.to_csv(output_tsv, sep="\t", index=False)

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()
        ap.add_argument("--cojo-dir", required=True)
        ap.add_argument("--manifest", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        build_cojo_table(Path(a.cojo_dir), Path(a.manifest), Path(a.out))
    ```

    """Assemble master_table.tsv from all Wave 3-4 outputs per RESEARCH §16."""
    from pathlib import Path
    import pandas as pd, glob, json, argparse

    KNOWN_OVERLAP_PAIRS = {
        ("hypertension", "gbmi_eur"): "GBMI-EUR includes UKBB; overlaps Evangelou 2018 HTN discovery",
        ("bmi", "gbmi_eur"): "GBMI lacks BMI; GIANT Yengo substitute overlaps UKBB → GBMI-EUR if BMI used",
        ("stroke", "bbj"): "MEGASTROKE any-stroke may overlap BBJ; BBJ uses ischemic-only (narrower)",
    }

    FINNISH_FOUNDER_FLAG_TRAITS = {"t2d", "hypertension"}  # flag low-MAF signals per RESEARCH pitfall #6

    def assemble_master_table(
        manifest_tsv: Path, fiqt_tsv: Path,
        per_cohort_dir: Path, coloc_dir: Path, meta_tsv: Path,
        output_tsv: Path,
    ) -> pd.DataFrame:
        manifest = pd.read_csv(manifest_tsv, sep="\t")
        fiqt = pd.read_csv(fiqt_tsv, sep="\t")
        meta = pd.read_csv(meta_tsv, sep="\t") if meta_tsv.exists() else pd.DataFrame()

        # Base row per signal
        base_cols = ["signal_id","signal_class","discovery_trait","discovery_ancestry","region",
                     "lead_snp","gene_assignment","tissue_or_celltype","qtl_source"]
        signals = manifest.drop_duplicates("signal_id")[["signal_id","signal_class","discovery_trait","discovery_ancestry","region"]].copy()
        signals = signals.merge(
            fiqt.rename(columns={"rsid":"lead_snp","beta":"beta_discovery_raw","se":"se_discovery_raw",
                                 "beta_FIQT":"beta_discovery_FIQT","se_FIQT":"se_discovery_FIQT"}),
            on="signal_id", how="left",
        )
        signals["p_discovery_raw"] = None  # filled from original sumstats if desired

        # Per-cohort blocks
        cohorts = ["finngen_r12", "gbmi_eur", "gbmi_afr", "mvp_eur", "mvp_afr"]
        for cohort in cohorts:
            pc_file = per_cohort_dir / f"{cohort}.tsv"
            if not pc_file.exists():
                # Emit empty cohort block so schema consistent
                for suffix in ["beta_replication","se_replication","p_replication","eaf_replication",
                               "power_posthoc","replicated_bonferroni",
                               "pph4_replication","replicated_pph4_0.5","replicated_pph4_0.7",
                               "replicated_pph4_0.8","replicated_pph4_0.9","replicated_joint_0.8"]:
                    signals[f"{cohort}_{suffix}"] = None
                continue
            pc = pd.read_csv(pc_file, sep="\t")
            # rename all columns with cohort prefix
            pc_renamed = pc.rename(columns={c: f"{cohort}_{c}" for c in pc.columns if c not in ("signal_id",)})
            signals = signals.merge(pc_renamed, on="signal_id", how="left")

        # Meta block
        if not meta.empty:
            signals = signals.merge(meta, on="signal_id", how="left")

        # Flags
        signals["sample_overlap_flag"] = signals.apply(
            lambda r: KNOWN_OVERLAP_PAIRS.get((r["discovery_trait"], "gbmi_eur"), None), axis=1
        )
        signals["low_maf_founder_flag"] = signals["discovery_trait"].isin(FINNISH_FOUNDER_FLAG_TRAITS)
        signals["notes"] = None

        signals.to_csv(output_tsv, sep="\t", index=False)
        return signals

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()
        ap.add_argument("--manifest", required=True)
        ap.add_argument("--fiqt", required=True)
        ap.add_argument("--per-cohort-dir", required=True)
        ap.add_argument("--coloc-dir", required=True)
        ap.add_argument("--meta", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        assemble_master_table(Path(a.manifest), Path(a.fiqt), Path(a.per_cohort_dir),
                              Path(a.coloc_dir), Path(a.meta), Path(a.out))
    ```

    """Tier A+B cross-ancestry generalization panel (BBJ-EAS column only, D-05c)."""
    from pathlib import Path
    import pandas as pd, argparse

    def build_bbj_generalization(manifest_tsv: Path, bbj_cohort_tsv: Path, output_tsv: Path):
        manifest = pd.read_csv(manifest_tsv, sep="\t")
        # D-05c enforcement: only Tier A+B triples, never credible-set SNPs
        tier_ab = manifest[manifest["signal_class"].isin(["tier_A_triple", "tier_B_triple"])]

        bbj = pd.read_csv(bbj_cohort_tsv, sep="\t") if bbj_cohort_tsv.exists() else pd.DataFrame()
        out = tier_ab_bbj.merge(bbj, on="signal_id", how="left", suffixes=("","_bbj"))
        out["is_generalization"] = True  # explicit methods label (D-05c)
        out["framing_note"] = "Cross-ancestry generalization evidence, NOT independent replication (D-05c)"
        out.to_csv(output_tsv, sep="\t", index=False)

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()

        ap.add_argument("--bbj-cohort", required=True)
        ap.add_argument("--out", required=True)

    """Leave-one-cohort-out IVW meta per signal (RESEARCH §16 recommendation)."""
    from pathlib import Path
    import pandas as pd, numpy as np, argparse

    def loco_meta(per_cohort_long: pd.DataFrame) -> pd.DataFrame:
        """per_cohort_long has columns: signal_id, cohort, cohort_ancestry, beta_replication, se_replication."""
        rows = []
        for sig, grp in per_cohort_long.groupby("signal_id"):
            ancestry = grp["cohort_ancestry"].iloc[0]
            matched = grp[(grp["cohort_ancestry"]==ancestry) & grp["beta_replication"].notna() & (grp["se_replication"]>0)]
            if len(matched) < 2:
                continue
            for i, hold_out in matched.iterrows():
                rest = matched.drop(i)
                if len(rest) < 2: continue
                w = 1 / rest["se_replication"]**2
                beta_loco = (w * rest["beta_replication"]).sum() / w.sum()
                se_loco = np.sqrt(1 / w.sum())
                rows.append({
                    "signal_id": sig,
                    "held_out_cohort": hold_out["cohort"],
                    "held_out_beta": hold_out["beta_replication"],
                    "held_out_se": hold_out["se_replication"],
                    "loco_meta_beta": beta_loco,
                    "loco_meta_se": se_loco,
                    "loco_n_cohorts": len(rest),
                })
        return pd.DataFrame(rows)

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()
        ap.add_argument("--per-cohort", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        df = pd.read_csv(a.per_cohort, sep="\t")
        loco_meta(df).to_csv(a.out, sep="\t", index=False)
    ```

    # Phase 9 Replication Methods

    ## Cohort portfolio (D-01)
    We assessed replication of Phase 1 credible-set lead SNPs and Phase 2 Tier A+B gene-tissue-trait
    triples across four independent cohorts: GBMI (Zhou et al. 2022 Cell Genomics; cross-biobank meta),
    FinnGen R12 (Kurki et al. 2023 Nature), MVP phs001672 (dbGaP open-access), and Biobank Japan
    hum0197-v3 (Sakaue et al. 2021 Nat Genet). All four are open-access per project data_access.md.

    ## Signal taxonomy (D-02)
    Two signal classes entered replication: (1) Phase 1 credible-set lead SNPs per trait × ancestry × region,
    (2) Phase 2 Tier A+B gene-tissue-trait triples. Tier C exploratory signals were excluded (D-02b).

    ## Joint replication criterion (D-03)
    A signal was declared replicated in a given cohort if and only if both conditions held:
    (i) same-direction β with p < Bonferroni threshold α = 0.05 / N_signals_tested_in_cohort;
    (ii) coloc.susie(discovery_fit, replication_fit) PP.H4 ≥ threshold, with sweep over {0.5, 0.7, 0.8, 0.9}.

    ## Effect-size correction (D-04)
    We applied FIQT (Bigdeli et al. 2016 Bioinformatics) empirical-Bayes winner's-curse shrinkage
    to discovery β̂ using the `winnerscurse` R package (Forde et al. 2023 Bioinformatics), pinned to
    a specific commit SHA for reproducibility. The master replication table reports four effect-size
    columns per signal × cohort: discovery raw, discovery FIQT-corrected, replication, and IVW meta.

    ## Ancestry-matched panels (D-05)
    EUR discoveries were replicated in FinnGen + GBMI-EUR + MVP-EUR. AFR discoveries were replicated
    in MVP-AFR + GBMI-AFR. For Tier A+B signals only, BBJ-EAS was used as a cross-ancestry
    **generalization** panel (explicitly NOT a replication cohort); signals that fail in BBJ-EAS
    are reported as ancestry-specific, not as failed replications.

    ## COJO sensitivity caveat (D-04c, RESEARCH gotcha #1)
    As orthogonal sensitivity we ran GCTA-COJO conditional+joint analysis at complex loci using
    1000 Genomes Phase 3 EUR (N=503) and AFR (N=661) LD references. Both panels are below GCTA's
    recommended N≥4000 threshold; we flag COJO as tier-2 supplementary evidence and do not treat
    COJO-failed loci as primary non-replications. The primary joint criterion uses coloc.susie
    re-estimation, not COJO.

    ## Stroke endpoint heterogeneity (RESEARCH gotcha #3)
    The four cohorts use different stroke definitions:
    - FinnGen R12 primary = `I9_STR_EXH` (exhaustive any-stroke union); sensitivity = `I9_STR`
    - BBJ = `IS` (ischemic-only)
    - MVP: see supplementary (subset verified from dbGaP phs001672 inventory)
    - MEGASTROKE (Phase 1 discovery) = any-stroke
    We adopted ischemic-only as the primary stroke endpoint across cohorts (consistent at BBJ +
    FinnGen `I9_STR`) and report `I9_STR_EXH` as sensitivity to document discovery-definition breadth.

    ## Outputs
    - `results/replication/master_table.tsv` — full replication matrix
    - `results/replication/cross_ancestry_generalization_tier_ab.tsv` — BBJ-EAS panel
    - `results/replication/cojo_sensitivity.tsv` — supplementary
    - `results/replication/replication_holdout_supplementary.tsv` — leave-one-cohort-out IVW meta
