    ```python
    """Build the Phase 9 replication manifest — signal × cohort crossmap.
    Consumes:
      - Phase 1 credible-set summary (finemap_tier*.tsv or .fit.rds listing)
      - Phase 2 tier_assignments.tsv (filter Tier A+B only)
      - config/replication_cohorts.yaml (panels + cohort trait availability)
    Emits:
      - data/processed/replication/manifest.tsv
    """
    from pathlib import Path
    import pandas as pd, yaml, sys, argparse

    PHASE1_FIT_GLOB = "results/fine_mapping/{trait}_{ancestry}_{region}.fit.rds"

    def _cohort_trait_available(cohort_cfg: dict, trait: str) -> bool:
        t = cohort_cfg.get("traits", {}).get(trait)
        if t is None: return False
        if isinstance(t, dict) and t.get("status") in ("EXCLUDED", "NOT_RELEASED_AS_OF_2026-04"):
            return False
        return True

    def _resolve_rep_path(cohort_name: str, trait: str, ancestry: str, cfg: dict) -> str:
        """Resolve expected path to harmonized sumstats (Plan 09-02 output)."""
        # Canonical output pattern: data/processed/replication/harmonized_grch37/{cohort}/{trait}[_{ancestry}].tsv.gz
        base = "data/processed/replication/harmonized_grch37"
        if cohort_name == "finngen_r12":
            endpoint = cfg["cohorts"]["finngen_r12"]["traits"][trait]["endpoint"]
            return f"{base}/finngen_r12/{trait}_{endpoint}.tsv.gz"
        if cohort_name.startswith("gbmi_"):
            anc = cohort_name.split("_")[1]
            return f"{base}/gbmi/{trait}_{anc}.tsv.gz"
        if cohort_name.startswith("mvp_"):
            anc = cohort_name.split("_")[1]
            pha = cfg["cohorts"]["mvp_phs001672"]["traits"][trait][anc]["pha"]
            return f"{base}/mvp/{pha}.tsv.gz"
        if cohort_name == "bbj":
            code = cfg["cohorts"]["bbj_hum0197_v3"]["traits"][trait]["trait_code"]
            return f"{base}/bbj/{trait}_{code}.tsv.gz"
        raise ValueError(f"Unknown cohort {cohort_name}")

    def _ld_panel_for(cohort_name: str, cfg: dict) -> str:
        """Route per-cohort LD panel per D-08."""
        anc_to_ld = cfg["ld_panels"]
        if cohort_name in ("finngen_r12",) or cohort_name.endswith("_eur"):
            return anc_to_ld["EUR"]
        if cohort_name.endswith("_afr"):
            return anc_to_ld["AFR"]
        if cohort_name == "bbj":
            return anc_to_ld["EAS"]
        raise ValueError(cohort_name)

    def build_manifest(
        credset_tsv: Path, tier_assignments_tsv: Path, config_yaml: Path, out_tsv: Path,
    ) -> pd.DataFrame:
        cfg = yaml.safe_load(open(config_yaml))
        credset = pd.read_csv(credset_tsv, sep="\t") if credset_tsv.exists() else pd.DataFrame()
        tiers = pd.read_csv(tier_assignments_tsv, sep="\t") if tier_assignments_tsv.exists() else pd.DataFrame()
        tier_ab = tiers[tiers["tier"].isin(["Tier A", "Tier B"])].copy()  # D-02b exclude Tier C

        rows = []
        panels = cfg["panels"]

        # Layer 1: credible-set SNPs
        for _, r in credset.iterrows():
            disc_anc = r["ancestry"].upper()
            panel_key = f"primary_{disc_anc.lower()}"
            if panel_key not in panels: continue
            for cohort_name in panels[panel_key]["cohorts"]:
                cohort_root = cohort_name.split("_")[0] + ("_" + cohort_name.split("_")[1] if "_" in cohort_name else "")
                # normalize: mvp_eur → cohorts['mvp_phs001672']
                cohort_cfg = cfg["cohorts"][
                    "mvp_phs001672" if cohort_name.startswith("mvp") else
                    ("gbmi" if cohort_name.startswith("gbmi") else
                     ("finngen_r12" if cohort_name == "finngen_r12" else "bbj_hum0197_v3"))
                ]
                if not _cohort_trait_available(cohort_cfg, r["trait"]): continue
                rows.append({
                    "signal_id": f"{r['trait']}_{disc_anc}_{r['region']}_{r['lead_snp']}",
                    "signal_class": "credible_set_SNP",
                    "discovery_trait": r["trait"],
                    "discovery_ancestry": disc_anc,
                    "region": r["region"],
                    "lead_snp": r["lead_snp"],
                    "cohort": cohort_name,
                    "cohort_ancestry": disc_anc,  # matched
                    "discovery_fit_path": PHASE1_FIT_GLOB.format(trait=r["trait"], ancestry=disc_anc, region=r["region"]),
                    "replication_sumstats_path": _resolve_rep_path(cohort_name, r["trait"], disc_anc.lower(), cfg),
                    "ld_panel": _ld_panel_for(cohort_name, cfg),
                })

        # Layer 2: Tier A+B triples — same primary panels PLUS BBJ (generalization)
        for _, r in tier_ab.iterrows():
            disc_anc = r["gwas_ancestry"].upper()
            # Primary ancestry-matched cohorts
            panel_key = f"primary_{disc_anc.lower()}"
            target_cohorts = list(panels.get(panel_key, {}).get("cohorts", []))
            # Generalization: add BBJ ONLY for Tier A+B (D-05c signal_scope: tier_ab_only)
            if disc_anc == "EUR":
                target_cohorts.append("bbj")
            for cohort_name in target_cohorts:
                cohort_cfg_name = (
                    "mvp_phs001672" if cohort_name.startswith("mvp") else
                    ("gbmi" if cohort_name.startswith("gbmi") else
                     ("finngen_r12" if cohort_name == "finngen_r12" else "bbj_hum0197_v3"))
                )
                cohort_cfg = cfg["cohorts"][cohort_cfg_name]
                if not _cohort_trait_available(cohort_cfg, r["gwas_trait"]): continue
                rep_anc = "EAS" if cohort_name == "bbj" else disc_anc
                rows.append({
                    "signal_id": f"{r['gwas_trait']}_{disc_anc}_{r['region']}_tier{r['tier'][-1]}_{r['gene_id']}",
                    "signal_class": f"tier_{r['tier'][-1]}_triple",
                    "discovery_trait": r["gwas_trait"],
                    "discovery_ancestry": disc_anc,
                    "region": r["region"],
                    "gene_id": r["gene_id"],
                    "tissue": r.get("tissue"),
                    "qtl_source": r.get("qtl_source"),
                    "cohort": cohort_name,
                    "cohort_ancestry": rep_anc,
                    "is_generalization": (cohort_name == "bbj"),
                    "discovery_fit_path": PHASE1_FIT_GLOB.format(trait=r["gwas_trait"], ancestry=disc_anc, region=r["region"]),
                    "replication_sumstats_path": _resolve_rep_path(cohort_name, r["gwas_trait"], rep_anc.lower(), cfg),
                    "ld_panel": _ld_panel_for(cohort_name, cfg),
                })

        df = pd.DataFrame(rows)
        out_tsv.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(out_tsv, sep="\t", index=False)
        return df

    if __name__ == "__main__":
        ap = argparse.ArgumentParser()
        ap.add_argument("--credset", required=True)
        ap.add_argument("--tiers", required=True)
        ap.add_argument("--config", required=True)
        ap.add_argument("--out", required=True)
        a = ap.parse_args()
        df = build_manifest(Path(a.credset), Path(a.tiers), Path(a.config), Path(a.out))
        print(f"manifest rows: {len(df)}; classes: {df['signal_class'].value_counts().to_dict()}")
    ```

    ```r
    #!/usr/bin/env Rscript
    # FIQT winner's-curse correction wrapper (D-04a).
    # Input: TSV with columns (rsid, beta, se, n) — from discovery signals
    # Output: TSV with added columns beta_FIQT, se_FIQT

    suppressPackageStartupMessages({
      if (!requireNamespace("winnerscurse", quietly=TRUE)) {
        if (!requireNamespace("remotes", quietly=TRUE)) install.packages("remotes", repos="https://cloud.r-project.org")
        remotes::install_github("amandaforde/winnerscurse", upgrade="never")
      }
      library(winnerscurse)
      library(data.table)
    })

    apply_fiqt <- function(df) {
      # winnerscurse::FDR_IQT expects data frame with columns rsid, beta, se, n
      required <- c("rsid", "beta", "se", "n")
      stopifnot(all(required %in% names(df)))
      res <- winnerscurse::FDR_IQT(summary_data = df, min_pval = 1e-300)
      # FDR_IQT returns a data frame with 'beta.FIQT' (or similar) column
      # Normalize column name
      fiqt_col <- grep("^beta.*FIQT|^FIQT", names(res), value = TRUE, ignore.case = TRUE)[1]
      if (is.na(fiqt_col)) stop("winnerscurse::FDR_IQT did not return a FIQT beta column")
      df$beta_FIQT <- res[[fiqt_col]]
      # SE shrinkage (winnerscurse may or may not provide shrunken SE directly)
      se_col <- grep("^se.*FIQT|^FIQT.*se", names(res), value = TRUE, ignore.case = TRUE)[1]
      df$se_FIQT <- if (!is.na(se_col)) res[[se_col]] else df$se  # fallback: SE unchanged
      df
    }

    if (!interactive() && sys.nframe() == 0L) {
      args <- commandArgs(trailingOnly = TRUE)
      stopifnot(length(args) >= 2)
      input <- args[1]; output <- args[2]
      df <- fread(input)
      setnames(df, tolower(names(df)))
      res <- apply_fiqt(df)
      fwrite(res, output, sep = "\t")
      message(sprintf("FIQT applied to %d signals; output: %s", nrow(res), output))
    }
    ```

    ```r
    #!/usr/bin/env Rscript
    # Fit SuSiE-RSS on replication cohort sumstats per region.
    # Reuses Phase 1 config/susie_policy.yaml (D-08 reuse, not fork).

    suppressPackageStartupMessages({
      library(susieR); library(coloc); library(data.table); library(yaml)
    })

    # Policy loader (reuse Phase 1 convention)
    load_policy <- function(path) yaml::read_yaml(path)

    # Main entrypoint (called from Snakemake or CLI)
    fit_replication_susie <- function(
      sumstats_tsv, ld_rds, output_rds, region,
      trait_type = c("cc","quant"), case_n = NULL, ctrl_n = NULL, total_n,
      policy_path = "config/susie_policy.yaml"
    ) {
      trait_type <- match.arg(trait_type)
      policy <- load_policy(policy_path)
      L <- policy$default$L %||% 10
      coverage <- policy$default$coverage %||% 0.95
      min_abs_corr <- policy$default$min_abs_corr %||% 0.5

      sumstats <- fread(sumstats_tsv)
      # Subset to region (chr:start-end)
      parts <- strsplit(gsub("chr", "", region), "[:-]")[[1]]
      sumstats <- sumstats[CHR == as.integer(parts[1]) & BP >= as.integer(parts[2]) & BP <= as.integer(parts[3])]
      if (nrow(sumstats) < 50) stop(sprintf("region %s has < 50 SNPs after filter", region))

      ld <- readRDS(ld_rds)
      # Align sumstats to LD matrix rows (assume SNP column matches LD rownames)
      common <- intersect(sumstats$SNP, rownames(ld))
      sumstats <- sumstats[SNP %in% common]
      ld <- ld[common, common]

      D <- list(
        beta = sumstats$BETA, varbeta = sumstats$SE^2,
        snp = sumstats$SNP, position = sumstats$BP,
        type = if (trait_type == "cc") "cc" else "quant",
        LD = ld, N = total_n, MAF = pmin(sumstats$EAF, 1 - sumstats$EAF)
      )
      if (trait_type == "cc") D$s <- case_n / (case_n + ctrl_n)

      # Retry ladder (reuse Phase 1 pattern — simplified: primary call + min_abs_corr fallback)
      fit <- tryCatch(
        coloc::runsusie(D, suffix = 2, L = L, coverage = coverage, min_abs_corr = min_abs_corr),
        error = function(e) {
          message("primary runsusie failed: ", e$message, " — retrying with min_abs_corr=0.1")
          coloc::runsusie(D, suffix = 2, L = L, coverage = coverage, min_abs_corr = 0.1)
        }
      )
      saveRDS(fit, output_rds)
      invisible(fit)
    }

    # Snakemake CLI
    if (!interactive() && sys.nframe() == 0L) {
      args <- commandArgs(trailingOnly = TRUE)
      named <- strsplit(args, "=")
      nm <- setNames(sapply(named, `[`, 2), sapply(named, `[`, 1))
      fit_replication_susie(
        sumstats_tsv = nm[["sumstats"]],
        ld_rds = nm[["ld"]],
        output_rds = nm[["out"]],
        region = nm[["region"]],
        trait_type = nm[["type"]],
        case_n = as.integer(nm[["case_n"]] %||% 0),
        ctrl_n = as.integer(nm[["ctrl_n"]] %||% 0),
        total_n = as.integer(nm[["n"]]),
        policy_path = nm[["policy"]] %||% "config/susie_policy.yaml"
      )
    }

    `%||%` <- function(a, b) if (is.null(a) || is.na(a) || identical(a, "")) b else a
    ```
