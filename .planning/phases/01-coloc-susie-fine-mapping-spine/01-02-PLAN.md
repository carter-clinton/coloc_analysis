    ```python
    #!/usr/bin/env python3
    """Download UKBB-LD tiled EUR panel (Weissbrod 2020) from AWS Open Data Registry
    and extract per-curated-region LD submatrices.

    Bucket: s3://broad-alkesgroup-ukbb-ld/UKBB_LD/
    License: Open Data Registry (public, no DUA)
    Reference: Weissbrod et al. 2020, Nature Genetics

    T-1-02 mitigation: SHA256 checksum verification on each tile.
    T-1-04 mitigation: HLA_6p21 gets ld_source='ukbb_ld_tiled_block_diagonal' flag.
    """
    import argparse, hashlib, json, sys
    from pathlib import Path
    import boto3
    from botocore import UNSIGNED
    from botocore.client import Config
    import numpy as np
    import pandas as pd
    import scipy.linalg
    import pyreadr  # or subprocess to Rscript for .rds writing

    BUCKET = "broad-alkesgroup-ukbb-ld"
    PREFIX = "UKBB_LD/"

    def make_s3_client():
        return boto3.client("s3", config=Config(signature_version=UNSIGNED))

    def list_tiles(s3, chrom):
        paginator = s3.get_paginator("list_objects_v2")
        tiles = []
        for page in paginator.paginate(Bucket=BUCKET, Prefix=f"{PREFIX}chr{chrom}_"):
            for obj in page.get("Contents", []):
                key = obj["Key"]
                if key.endswith(".npz"):
                    stem = Path(key).stem  # chr{N}_{start}_{end}
                    parts = stem.split("_")
                    try:
                        start = int(parts[1]); end = int(parts[2])
                    except (IndexError, ValueError):
                        continue
                    tiles.append((start, end, key))
        return sorted(tiles)

    def tiles_for_region(tiles, start, end):
        return [(s, e, k) for s, e, k in tiles if e >= start and s <= end]

    def sha256_file(path):
        h = hashlib.sha256()
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(1 << 20), b""):
                h.update(chunk)
        return h.hexdigest()

    def download_tile(s3, key, local_path):
        Path(local_path).parent.mkdir(parents=True, exist_ok=True)
        if not Path(local_path).exists():
            s3.download_file(BUCKET, key, local_path)
        return sha256_file(local_path)

    def reconstruct_symmetric(npz, key_name="R"):
        """Reconstruct symmetric LD matrix from upper-triangle flat array.
        Actual key name verified at Wave 0 preflight (wave2a_preflight.log).
        """
        R_upper = npz[key_name]
        n = int((np.sqrt(1 + 8 * len(R_upper)) - 1) / 2)
        assert n * (n + 1) // 2 == len(R_upper), f"Expected triangular size, got n={n} len={len(R_upper)}"
        R = np.zeros((n, n), dtype=np.float32)
        iu = np.triu_indices(n)
        R[iu] = R_upper
        R = R + R.T - np.diag(np.diag(R))
        return R

    def extract_region(s3, chrom, region_id, region_start, region_end, out_dir, scratch_dir, npz_key_name="R"):
        tiles = list_tiles(s3, chrom)
        overlapping = tiles_for_region(tiles, region_start, region_end)
        if not overlapping:
            raise RuntimeError(f"No UKBB-LD tiles overlap {chrom}:{region_start}-{region_end}")

        sha_manifest = {}
        if len(overlapping) == 1:
            tile_start, tile_end, key = overlapping[0]
            npz_local = Path(scratch_dir) / Path(key).name
            var_local = npz_local.with_suffix(".gz")
            sha_manifest[key] = download_tile(s3, key, str(npz_local))
            sha_manifest[key.replace(".npz", ".gz")] = download_tile(s3, key.replace(".npz", ".gz"), str(var_local))

            arr = np.load(npz_local)
            R_full = reconstruct_symmetric(arr, key_name=npz_key_name)
            variants = pd.read_csv(var_local, sep="\t", compression="gzip")
            # normalize position column name
            pos_col = next((c for c in ["position","pos","POS","bp"] if c in variants.columns), None)
            assert pos_col, f"No position column in variants TSV; columns: {list(variants.columns)}"
            mask = (variants[pos_col] >= region_start) & (variants[pos_col] <= region_end)
            idx = mask.values
            R_region = R_full[np.ix_(idx, idx)]
            variants_region = variants[mask].reset_index(drop=True)
            ld_source = "ukbb_ld_tiled"
        else:
            # Multi-tile block-diagonal (HLA_6p21)
            blocks_R, blocks_v = [], []
            tile_keys = []
            for tile_start, tile_end, key in overlapping:
                npz_local = Path(scratch_dir) / Path(key).name
                var_local = npz_local.with_suffix(".gz")
                sha_manifest[key] = download_tile(s3, key, str(npz_local))
                sha_manifest[key.replace(".npz", ".gz")] = download_tile(s3, key.replace(".npz", ".gz"), str(var_local))
                arr = np.load(npz_local)
                R_tile = reconstruct_symmetric(arr, key_name=npz_key_name)
                var_tile = pd.read_csv(var_local, sep="\t", compression="gzip")
                pos_col = next((c for c in ["position","pos","POS","bp"] if c in var_tile.columns), None)
                mask = (var_tile[pos_col] >= region_start) & (var_tile[pos_col] <= region_end)
                idx = mask.values
                if idx.sum() == 0:
                    continue
                blocks_R.append(R_tile[np.ix_(idx, idx)])
                blocks_v.append(var_tile[mask].reset_index(drop=True))
                tile_keys.append(key)
            R_region = scipy.linalg.block_diag(*blocks_R)
            variants_region = pd.concat(blocks_v, ignore_index=True)
            ld_source = "ukbb_ld_tiled_block_diagonal"

        # Write .rds
        out_rds = Path(out_dir) / f"{region_id}.rds"
        out_rds.parent.mkdir(parents=True, exist_ok=True)
        # pyreadr write: wrap as a list via data.frame for R consumption
        pyreadr.write_rds(str(out_rds), {"R": pd.DataFrame(R_region),
                                         "variants": variants_region})
        meta = {
            "region_id": region_id, "chr": chrom,
            "start": region_start, "end": region_end,
            "n_variants": int(len(variants_region)),
            "ld_source": ld_source,
            "tile_keys": tile_keys if len(overlapping) > 1 else [overlapping[0][2]],
            "sha256": sha_manifest,
            "source_paper": "Weissbrod 2020 Nat Genet (UKBB-LD)",
        }
        out_rds.with_suffix(".meta.json").write_text(json.dumps(meta, indent=2))
        return ld_source

    def main():
        ap = argparse.ArgumentParser()
        ap.add_argument("--regions-csv", required=True)
        ap.add_argument("--out-dir", required=True)
        ap.add_argument("--scratch-dir", default="/rs1/scratch/ukbb_ld")
        ap.add_argument("--npz-key-name", default="R", help="NPZ internal key — set from preflight log")
        ap.add_argument("--region-ids", nargs="*", default=None, help="Subset to these region_ids (default: all)")
        args = ap.parse_args()

        s3 = make_s3_client()
        regions = pd.read_csv(args.regions_csv)
        if args.region_ids:
            regions = regions[regions["region_id"].isin(args.region_ids)]
        for _, row in regions.iterrows():
            ld_source = extract_region(
                s3, str(row["chr"]), row["region_id"],
                int(row["start"]), int(row["end"]),
                args.out_dir, args.scratch_dir,
                npz_key_name=args.npz_key_name,
            )
            print(f"[{row['region_id']}] ld_source={ld_source}")

    if __name__ == "__main__":
        main()
    ```
