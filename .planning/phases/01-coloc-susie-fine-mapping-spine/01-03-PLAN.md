    ```python
    #!/usr/bin/env python3
    """Build HGDP+1kG AFR LD panel from gnomAD v3.1.2 phased BCFs (anonymous HTTPS).

    Source: https://gnomad.broadinstitute.org/news/2021-10-gnomad-v3-1-2-minor-release/
    License: Public (CC-BY), no DUA
    Reference: atgu/hgdp_tgp

    T-1-02 mitigation: SHA256 computed on BCF slice + metadata file; recorded in sidecar .meta.json.
    T-1-04 mitigation: ld_source field set to 'hgdp_1kg_v3_1_2'.
    """
    import argparse, hashlib, json, subprocess, sys
    from pathlib import Path
    import pandas as pd

    META_URL = "https://storage.googleapis.com/gcp-public-data--gnomad/release/3.1.2/pca/gnomad.v3.1.2.hgdp_tgp_meta.tsv"
    BCF_URL_TPL = "https://storage.googleapis.com/gcp-public-data--gnomad/resources/hgdp_1kg/phased_haplotypes_v2/{fname}"

    def sha256_file(path):
        h = hashlib.sha256()
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(1 << 20), b""):
                h.update(chunk)
        return h.hexdigest()

    def download_metadata(meta_path):
        if not meta_path.exists():
            subprocess.run(["curl", "-sL", META_URL, "-o", str(meta_path)], check=True)
        return pd.read_csv(meta_path, sep="\t", low_memory=False)

    def extract_afr_samples(meta_df, reg_col, s_col, out_path, bcf_sample_list=None):
        afr = meta_df[meta_df[reg_col] == "AFR"]
        samples = afr[s_col].astype(str).tolist()
        if bcf_sample_list:
            # Reconcile against actual BCF header (Pitfall 3 — prefix mismatch)
            bcf_samples = set(Path(bcf_sample_list).read_text().split())
            reconciled = [s for s in samples if s in bcf_samples]
            if not reconciled:
                # Try common prefix variants
                for prefix in ["HGDP_", "1KG_", ""]:
                    cand = [f"{prefix}{s}" for s in samples]
                    reconciled = [s for s in cand if s in bcf_samples]
                    if reconciled:
                        break
            samples = reconciled
        Path(out_path).write_text("\n".join(samples) + "\n")
        return len(samples)

    def stream_bcf_slice(bcf_url, samples_file, chrom, start, end, out_bcf, maf=0.01):
        cmd = ["bcftools", "view",
               "-S", str(samples_file),
               "-r", f"{chrom}:{start}-{end}",
               "--min-af", str(maf),
               "-Ob", "-o", str(out_bcf),
               bcf_url]
        subprocess.run(cmd, check=True)

    def plink2_ld(bcf_path, out_prefix):
        cmd = ["plink2",
               "--bcf", str(bcf_path),
               "--r2-phased", "square",
               "--out", str(out_prefix)]
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            # Fallback to --r2 if --r2-phased unavailable
            cmd = ["plink2", "--bcf", str(bcf_path), "--r2", "square", "--out", str(out_prefix)]
            subprocess.run(cmd, check=True)

    def convert_to_rds(plink_prefix, out_rds, region_id, chrom, start, end, sha_manifest, sample_count):
        subprocess.run([
            "Rscript", "src/snakemake/scripts/plink_ld_to_rds.R",
            "--ld", f"{plink_prefix}.phased.vcor2",  # or .ld depending on plink version
            "--variants", f"{plink_prefix}.pvar",
            "--region-id", region_id,
            "--ancestry", "AFR",
            "--output", str(out_rds),
        ], check=True)
        meta = {
            "region_id": region_id, "chr": str(chrom),
            "start": int(start), "end": int(end),
            "ld_source": "hgdp_1kg_v3_1_2",
            "n_samples_afr": sample_count,
            "sha256": sha_manifest,
            "source_paper": "gnomAD v3.1.2 HGDP+1kG phased haplotypes v2",
        }
        Path(out_rds).with_suffix(".meta.json").write_text(json.dumps(meta, indent=2))

    def main():
        ap = argparse.ArgumentParser()
        ap.add_argument("--regions-csv", required=True)
        ap.add_argument("--out-dir", required=True)
        ap.add_argument("--scratch-dir", default="/rs1/scratch/hgdp_1kg")
        ap.add_argument("--bcf-fname-template", required=True,
                        help="e.g. hgdp1kgp_chr{chrom}.filtered.SNV_INDEL.phased.shapeit5.bcf — from preflight")
        ap.add_argument("--region-column", default="genetic_region")
        ap.add_argument("--sample-column", default="s")
        ap.add_argument("--region-ids", nargs="*", default=None,
                        help="Subset to these region_ids (Scope B pilot)")
        args = ap.parse_args()

        scratch = Path(args.scratch_dir); scratch.mkdir(parents=True, exist_ok=True)
        out_dir = Path(args.out_dir); out_dir.mkdir(parents=True, exist_ok=True)

        meta = download_metadata(scratch / "meta.tsv")
        afr_samples_file = scratch / "afr_samples.txt"
        sample_count = extract_afr_samples(meta, args.region_column, args.sample_column, afr_samples_file)
        print(f"[afr] {sample_count} samples extracted")

        regions = pd.read_csv(args.regions_csv)
        if args.region_ids:
            regions = regions[regions["region_id"].isin(args.region_ids)]

        for _, row in regions.iterrows():
            chrom = str(row["chr"]); rid = row["region_id"]
            start = int(row["start"]); end = int(row["end"])
            bcf_url = BCF_URL_TPL.format(fname=args.bcf_fname_template.format(chrom=chrom))
            slice_bcf = scratch / f"{rid}.bcf"
            stream_bcf_slice(bcf_url, afr_samples_file, chrom, start, end, slice_bcf)
            sha = {str(slice_bcf): sha256_file(slice_bcf)}
            prefix = scratch / f"ld_afr_{rid}"
            plink2_ld(slice_bcf, prefix)
            convert_to_rds(prefix, out_dir / f"{rid}.rds", rid, chrom, start, end, sha, sample_count)
            print(f"[{rid}] AFR LD built")

    if __name__ == "__main__":
        main()
    ```

    ```r
    #!/usr/bin/env Rscript
    # Convert plink2 phased-r2 text output to the legacy .rds shape
    # consumed by run_susie_rss.R load_ld_matrix.
    suppressPackageStartupMessages({ library(optparse); library(data.table) })

    opt <- parse_args(OptionParser(option_list = list(
      make_option("--ld", type="character"),
      make_option("--variants", type="character"),
      make_option("--region-id", dest="region_id", type="character"),
      make_option("--ancestry", type="character"),
      make_option("--output", type="character")
    )))

    R <- as.matrix(fread(opt$ld, header = FALSE))
    storage.mode(R) <- "numeric"
    R <- R + t(R) - diag(diag(R))  # ensure symmetric (plink writes upper or full square)

    # pvar is plink2's variant file: #CHROM ID POS REF ALT
    pv <- fread(opt$variants, skip = "#CHROM", header = TRUE)
    variants <- data.frame(
      SNP_ID = pv$ID,
      CHR = pv$`#CHROM`,
      POS = pv$POS,
      A1 = pv$REF,
      A2 = pv$ALT,
      stringsAsFactors = FALSE
    )
    stopifnot(nrow(R) == nrow(variants))

    ld_obj <- list(R = R, variants = variants,
                   ld_source = "hgdp_1kg_v3_1_2",
                   region_id = opt$region_id,
                   ancestry = opt$ancestry)
    saveRDS(ld_obj, opt$output)
    cat("[plink_ld_to_rds] wrote", opt$output, "with", nrow(R), "variants\n")
    ```
