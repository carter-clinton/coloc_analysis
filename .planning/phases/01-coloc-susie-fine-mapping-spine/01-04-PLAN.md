    #!/usr/bin/env Rscript
    # Phase 1 Wave 3 — coloc.susie replacement for the legacy run_coloc.R (coloc.abf).
    # Loads two .fit.rds files produced by run_susie_rss.R and calls coloc::coloc.susie.
    # Emits a JSON compatible with legacy downstream consumers:
    #   augment_coloc_summary.py, build_coloc_h4_reports.py, build_coloc_top_hits_table.py.
    # REQ-2 success criterion #4: coloc.susie replaces coloc.abf.
    # T-1-06: Uses coloc::coloc.susie dispatch on class("susie") objects (A6 branch per 01-01 summary).

    suppressPackageStartupMessages({
      library(optparse); library(coloc); library(susieR)
      library(jsonlite); library(data.table)
    })

    `%||%` <- function(x, y) if (!is.null(x)) x else y

    option_list <- list(
      make_option("--fit-a",   dest="fit_a",   type="character"),
      make_option("--fit-b",   dest="fit_b",   type="character"),
      make_option("--policy",  type="character", default="config/susie_policy.yaml"),
      make_option("--pair-id", dest="pair_id", type="character"),
      make_option("--manifest",type="character",
                  help="Path to coloc_manifest.tsv (needed to recover trait/ancestry/region from pair_id)"),
      make_option("--trait-a", dest="trait_a", type="character", default=NA_character_),
      make_option("--trait-b", dest="trait_b", type="character", default=NA_character_),
      make_option("--ancestry",type="character", default=NA_character_),
      make_option("--region",  type="character", default=NA_character_),
      make_option("--output",  type="character")
    )
    opt <- parse_args(OptionParser(option_list = option_list))

    stopifnot(file.exists(opt$fit_a), file.exists(opt$fit_b))
    fit_a <- readRDS(opt$fit_a)
    fit_b <- readRDS(opt$fit_b)

    # A6 safety: ensure class dispatch
    if (!inherits(fit_a, "susie")) class(fit_a) <- c("susie", class(fit_a))
    if (!inherits(fit_b, "susie")) class(fit_b) <- c("susie", class(fit_b))

    # Recover trait/ancestry/region from manifest if not provided on CLI
    if (!is.null(opt$manifest) && file.exists(opt$manifest)) {
      mf <- fread(opt$manifest)
      row <- mf[pair_id == opt$pair_id]
      if (nrow(row) == 1) {
        if (is.na(opt$trait_a))  opt$trait_a  <- row$trait_a
        if (is.na(opt$trait_b))  opt$trait_b  <- row$trait_b
        if (is.na(opt$ancestry)) opt$ancestry <- row$ancestry
        if (is.na(opt$region))   opt$region   <- row$region
      }
    }

    # Guard: empty-CS fits emit a no-signal JSON and exit cleanly
    n_cs_a <- length(fit_a$sets$cs %||% list())
    n_cs_b <- length(fit_b$sets$cs %||% list())
    if (n_cs_a == 0 || n_cs_b == 0) {
      empty <- list(
        pair_id = opt$pair_id,
        status  = "no_signal",
        trait_a = opt$trait_a, trait_b = opt$trait_b,
        ancestry = opt$ancestry, region = opt$region,
        base_region = opt$region,
        n_cs_a = n_cs_a, n_cs_b = n_cs_b,
        summary = list(
          nsnps = NA_integer_,
          PP.H0.abf = NA_real_, PP.H1.abf = NA_real_,
          PP.H2.abf = NA_real_, PP.H3.abf = NA_real_,
          PP.H4.abf = NA_real_
        ),
        susie_pairs = list(),
        n_pairs_total = 0
      )
      dir.create(dirname(opt$output), recursive = TRUE, showWarnings = FALSE)
      write_json(empty, opt$output, auto_unbox = TRUE, pretty = TRUE, na = "null")
      cat("[run_coloc_susie] no signal for", opt$pair_id, "— wrote empty JSON\n")
      quit(status = 0)
    }

    # Main call
    res <- coloc::coloc.susie(fit_a, fit_b)
    # res$summary is a data.table with one row per pairwise CS comparison:
    #   idx1, idx2, nsnps, hit1, hit2, PP.H0.abf..PP.H4.abf
    summary_dt <- as.data.table(res$summary)

    # Validate posterior sum (property check)
    pp_cols <- c("PP.H0.abf","PP.H1.abf","PP.H2.abf","PP.H3.abf","PP.H4.abf")
    if (all(pp_cols %in% names(summary_dt))) {
      row_sums <- rowSums(summary_dt[, ..pp_cols])
      if (any(abs(row_sums - 1.0) > 1e-4, na.rm = TRUE)) {
        warning("coloc.susie posterior sum deviation > 1e-4 detected in some rows")
      }
    }

    # Legacy compat layer (Pattern 6 Option A): best pairwise -> summary
    best_idx <- which.max(summary_dt$PP.H4.abf)
    best_row <- if (length(best_idx) == 1) as.list(summary_dt[best_idx]) else list()

    output <- list(
      pair_id = opt$pair_id,
      status  = "success",
      trait_a = opt$trait_a, trait_b = opt$trait_b,
      ancestry = opt$ancestry, region = opt$region,
      base_region = opt$region,
      summary = best_row,                      # legacy-compat: best pairwise row
      susie_pairs = as.list(summary_dt),       # new field: all pairwise rows
      n_pairs_total = nrow(summary_dt),
      n_cs_a = n_cs_a, n_cs_b = n_cs_b
    )

    dir.create(dirname(opt$output), recursive = TRUE, showWarnings = FALSE)
    write_json(output, opt$output, auto_unbox = TRUE, pretty = TRUE, na = "null")
    cat("[run_coloc_susie] wrote", opt$output, "with", nrow(summary_dt), "pairwise rows\n")
    ```

    """coloc.susie replacement rule (REQ-2 success criterion #4).

    Replaces the legacy run_coloc_pair rule in multitrait.smk:116 (coloc.abf).
    Depends on .fit.rds outputs from finemap.smk:run_finemap (Phase 1 Wave 1).
    Uses the legacy-compat JSON schema so downstream augment_coloc_summary.py,
    build_coloc_h4_reports.py, and build_coloc_top_hits_table.py continue working.
    """

    import os
    import sys

    PYTHON_BIN = sys.executable
    COLOC_SUSIE_DIR = os.path.join(MULTITRAIT_DIR, "coloc_susie")


    def _coloc_manifest_row(pair_id):
        """Resolve (trait_a, trait_b, ancestry, region) from the coloc_manifest TSV.
        The manifest is produced upstream by multitrait.smk:create_coloc_manifest.
        Row schema: pair_id, trait_a, trait_b, ancestry, region, ...
        """
        import pandas as pd
        manifest_path = os.path.join(MULTITRAIT_DIR, "coloc_manifest.tsv")
        if not os.path.exists(manifest_path):
            return None
        df = pd.read_csv(manifest_path, sep="\t")
        row = df[df["pair_id"] == pair_id]
        if len(row) != 1:
            return None
        return row.iloc[0].to_dict()


    def _fit_rds_for(trait, ancestry, region_safe):
        """Return the .fit.rds path produced by run_finemap for (trait, ancestry, region)."""
        return finemap_output("susie", trait, ancestry, region_safe).replace(".json", ".fit.rds")


    rule run_coloc_susie:
        input:
            manifest=os.path.join(MULTITRAIT_DIR, "coloc_manifest.tsv"),
            policy="config/susie_policy.yaml",
            fit_a=lambda wc: _fit_rds_for(
                _coloc_manifest_row(wc.pair_id)["trait_a"],
                _coloc_manifest_row(wc.pair_id)["ancestry"],
                _coloc_manifest_row(wc.pair_id)["region"],
            ),
            fit_b=lambda wc: _fit_rds_for(
                _coloc_manifest_row(wc.pair_id)["trait_b"],
                _coloc_manifest_row(wc.pair_id)["ancestry"],
                _coloc_manifest_row(wc.pair_id)["region"],
            ),
            script="src/snakemake/scripts/run_coloc_susie.R",
        output:
            json=os.path.join(COLOC_SUSIE_DIR, "{pair_id}.json"),
        conda:
            "envs/r_coloc.yml"
        threads: 1
        resources:
            mem_mb=config["resources"]["default_mem_mb"],
        shell:
            r"""
            mkdir -p $(dirname {output.json})
            Rscript {input.script} \
                --fit-a {input.fit_a} \
                --fit-b {input.fit_b} \
                --policy {input.policy} \
                --pair-id {wildcards.pair_id} \
                --manifest {input.manifest} \
                --output {output.json}
            """
    ```
