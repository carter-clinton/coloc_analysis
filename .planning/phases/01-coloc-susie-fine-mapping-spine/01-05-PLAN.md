    #!/usr/bin/env python3
    """Aggregate run_finemap JSON outputs into a single TSV for the QC dashboard.

    Reads all *.json under {FINEMAP_DIR}/susie/ (recursively), extracts D1/D2/D3
    diagnostic fields + ld_source + L_saturated + min_abs_corr_sweep summary,
    and writes a flat TSV keyed by (trait x ancestry x region).

    Phase 1 REQ-2 acceptance #5 — Per-locus fine-mapping QC report.
    T-1-04 mitigation: surface ld_source field so 'ukbb_ld_tiled_block_diagonal'
    is visible in the dashboard.
    """
    import argparse
    import json
    from pathlib import Path
    import pandas as pd

    def flatten_one(json_path):
        data = json.loads(Path(json_path).read_text())
        d1 = data.get("d1_zscore_sanity", {}) or {}
        d2 = data.get("d2_convergence", {}) or {}
        d3 = data.get("d3_ld_quality", {}) or {}
        sweep = {float(r["min_abs_corr"]): r for r in (data.get("min_abs_corr_sweep") or [])}

        row = {
            "region_id": data.get("region_id", ""),
            "trait": data.get("trait", ""),
            "ancestry": data.get("ancestry", ""),
            "status": data.get("status", ""),
            "convergence_status": data.get("convergence_status", ""),
            "L_used": data.get("L_used", None),
            "L_saturated": data.get("L_saturated", False),
            "ld_source": data.get("ld_source", ""),
            "n_variants": data.get("n_variants", None),
            # D1
            "ks_pvalue": d1.get("ks_pvalue", None),
            "max_abs_z": d1.get("max_abs_z", None),
            "lambda_gc": d1.get("lambda_gc", None),
            # D2
            "converged": d2.get("converged", None),
            "niter": d2.get("niter", None),
            "elbo_final": d2.get("elbo_final", None),
            # D3
            "kriging_n_outliers": d3.get("n_outliers", None),
            "kriging_max_logLR": d3.get("max_logLR", None),
            "kriging_lambda": d3.get("lambda", None),
            # D4 (via sweep)
            "n_CS_macor_0.1": (sweep.get(0.1) or {}).get("n_CS", 0),
            "n_CS_macor_0.5": (sweep.get(0.5) or {}).get("n_CS", 0),
            "n_CS_macor_0.9": (sweep.get(0.9) or {}).get("n_CS", 0),
        }
        return row

    def main():
        ap = argparse.ArgumentParser()
        ap.add_argument("--input-dir", required=True, help="e.g. results/finemap/susie")
        ap.add_argument("--output", required=True, help="TSV output path")
        args = ap.parse_args()

        input_dir = Path(args.input_dir)
        rows = []
        for jf in sorted(input_dir.rglob("*.json")):
            # Skip .fit.rds companions which aren't JSON
            if jf.suffix != ".json":
                continue
            try:
                rows.append(flatten_one(jf))
            except Exception as e:
                print(f"[warn] skipping {jf}: {e}")

        df = pd.DataFrame(rows)
        if df.empty:
            df = pd.DataFrame(columns=[
                "region_id","trait","ancestry","status","convergence_status",
                "L_used","L_saturated","ld_source","n_variants",
                "ks_pvalue","max_abs_z","lambda_gc",
                "converged","niter","elbo_final",
                "kriging_n_outliers","kriging_max_logLR","kriging_lambda",
                "n_CS_macor_0.1","n_CS_macor_0.5","n_CS_macor_0.9",
            ])
        df.to_csv(args.output, sep="\t", index=False)
        print(f"[aggregate] wrote {len(df)} rows to {args.output}")

    if __name__ == "__main__":
        main()
    ```

        for r in policy.get("complex_regions", {}).get("pre_specified", []):
            trait_by_region[r["region_id"]] = set(r.get("trait_list", []))

        known_mask = df["region_id"].isin(KNOWN_COMPLEX) & df.apply(
            lambda row: row["trait"] in trait_by_region.get(row["region_id"], set()),
            axis=1
        )
        known = df[known_mask].assign(row_group="known_complex")

        flagged_mask = ~known_mask & (
            (df["l_saturated"] == True) | (df["n_CS_macor_0.5"] >= 3)
        )
        flagged = df[flagged_mask].assign(row_group="data_flagged")

        sweep_df = pd.concat([known, flagged], ignore_index=True)
        sweep_df = sweep_df[sweep_cols].sort_values(
            ["row_group", "region_id", "trait", "ancestry"]
        )
        return sweep_df

    # Accept a second CLI arg --sweep-out
    if args.sweep_out:
        import yaml
        with open(args.policy) as f:
            policy = yaml.safe_load(f)
        sweep_df = build_sweep_table(df, policy)
        header_prefix = ""
        if sweep_df.empty:
            header_prefix = "## row_group_empty: both groups empty on current data\n"
        with open(args.sweep_out, "w") as f:
            f.write(header_prefix)
            sweep_df.to_csv(f, sep="\t", index=False)
    ```
