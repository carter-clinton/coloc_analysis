```python
import os
import csv
import yaml

# Test Snakefile -- uses same rules as production but with toy config
configfile: "tests/toy_3locus/config_test.yaml"

from snakemake.utils import validate
validate(config, "src/snakemake/schemas/pipeline.schema.yaml")

# ... (replicate the Snakefile header logic for trait/ancestry/region expansion
#      but scoped to toy data)

TRAITS = config["traits"]
ANCESTRIES = config["ancestries"]
TRAIT_ANCESTRIES = config.get("trait_ancestries", {})
HARMONIZED_DIR = config["paths"]["harmonized_sumstats"]

TRAIT_ANCESTRY_PAIRS = []
for trait in TRAITS:
    ancestries = TRAIT_ANCESTRIES.get(trait, ANCESTRIES)
    for anc in ancestries:
        TRAIT_ANCESTRY_PAIRS.append((trait, anc))

# Load toy regions
REGION_INFOS = []
REGION_SAFE_TO_ID = {}
REGION_METADATA = {}
with open(config["paths"]["regions_curated"], newline="") as fh:
    reader = csv.DictReader(fh)
    for row in reader:
        orig = row["region_id"]
        safe = orig.replace(".", "_").replace("/", "_")
        chrom = row["chr"].replace("chr", "")
        REGION_INFOS.append((orig, safe))
        REGION_SAFE_TO_ID[safe] = orig
        REGION_METADATA[safe] = {
            "chr": chrom,
            "start": int(float(row["start"])),
            "end": int(float(row["end"])),
        }

REGION_SAFE_IDS = [safe for _, safe in REGION_INFOS]

# Include the SAME production rules (D-04 -- refactor, not rewrite)
include: "src/snakemake/rules/sumstats.smk"
include: "src/snakemake/rules/regions.smk"
include: "src/snakemake/rules/qc.smk"

FINEMAP_METHODS = config.get("finemap", {}).get("methods", [])
if FINEMAP_METHODS:
    include: "src/snakemake/rules/finemap.smk"

if config.get("enable_ld_pipeline", False):
    include: "src/snakemake/rules/ld_reference.smk"

# Toy targets -- only harmonization, regions, LD, and finemap
# No multitrait, MR, or PGS in the smoke test
HARMONIZED_ALL = [
    os.path.join(HARMONIZED_DIR, f"{t}.{a}.tsv.bgz")
    for t, a in TRAIT_ANCESTRY_PAIRS
]

FINEMAP_DIR = config.get("finemap", {}).get("output_dir", "tests/toy_3locus/results/fine_mapping")

rule all:
    input:
        HARMONIZED_ALL,
        config["paths"]["regions_bed"],
```

IMPORTANT: The `include:` paths must be relative to the directory from which snakemake is invoked (project root), NOT relative to the Snakefile.test location. Since the invocation command is `snakemake --snakefile tests/toy_3locus/Snakefile.test` from project root, the includes should be `include: "src/snakemake/rules/sumstats.smk"` (same as production).

Create `scripts/subset_toy_loci.py` -- a utility script that extracts toy subsets from full sumstats files for the 3 test loci. This script is run ONCE after full data is downloaded (Phase 0 Track 0a or Phase 1 data preparation) to populate `tests/toy_3locus/data/`.

```python
#!/usr/bin/env python3
"""
Subset GWAS summary statistics to toy 3-locus regions for CI smoke testing.

Usage:
    python scripts/subset_toy_loci.py --config config/pipeline.yaml --test-config tests/toy_3locus/config_test.yaml

Reads full sumstats from paths in --config, subsets to regions defined in
tests/toy_3locus/data/regions_toy.csv, writes subsets to paths in --test-config.

Requires: tabix-indexed (.bgz + .tbi) harmonized sumstats.
"""

import argparse
import csv
import os
import subprocess
import sys
import yaml


TOY_LOCI = {
    "FTO_16q12_2":    {"chr": "16", "start": 53800000, "end": 54400000},
    "TCF7L2_10q25_2": {"chr": "10", "start": 114550000, "end": 115150000},
    "SH2B3_12q24_12": {"chr": "12", "start": 111400000, "end": 112000000},
}


def subset_sumstats(input_bgz, output_path, regions):
    """Extract rows within genomic regions from a bgzipped, tabix-indexed sumstats file."""
    # Get header
    header_cmd = ["bash", "-c", f"zcat '{input_bgz}' | head -1"]
    header = subprocess.run(header_cmd, capture_output=True, text=True, check=True)

    rows = []
    for name, region in regions.items():
        query = f"{region['chr']}:{region['start']}-{region['end']}"
        cmd = ["tabix", input_bgz, query]
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode == 0 and result.stdout.strip():
            rows.append(result.stdout.strip())

    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    with open(output_path, "w") as f:
        f.write(header.stdout)
        for block in rows:
            f.write(block + "\n")

    print(f"  Wrote {sum(len(b.split(chr(10))) for b in rows)} rows to {output_path}")


def main():
    parser = argparse.ArgumentParser(description="Subset sumstats for toy 3-locus smoke test")
    parser.add_argument("--config", default="config/pipeline.yaml", help="Production pipeline config")
    parser.add_argument("--test-config", default="tests/toy_3locus/config_test.yaml", help="Test config")
    parser.add_argument("--regions", default="tests/toy_3locus/data/regions_toy.csv", help="Toy regions CSV")
    args = parser.parse_args()

    cfg = yaml.safe_load(open(args.config))
    test_cfg = yaml.safe_load(open(args.test_config))

    harmonized_dir = cfg["paths"]["harmonized_sumstats"]
    test_raw_dir = test_cfg["paths"]["raw_sumstats"]

    # Subset each trait x ancestry pair
    for trait in test_cfg["traits"]:
        ancestries = test_cfg.get("trait_ancestries", {}).get(trait, test_cfg["ancestries"])
        for ancestry in ancestries:
            input_file = os.path.join(harmonized_dir, f"{trait}.{ancestry}.tsv.bgz")
            output_file = os.path.join(test_raw_dir, f"{trait}.{ancestry}.tsv")

            if not os.path.exists(input_file):
                print(f"  SKIP: {input_file} not found (data not yet downloaded)")
                continue

            print(f"  Subsetting {trait}.{ancestry}...")
            subset_sumstats(input_file, output_file, TOY_LOCI)

    print("Done. Populate tests/toy_3locus/data/ with the subsetted files.")


if __name__ == "__main__":
    main()
```

- For each trait x ancestry pair in the test config, extracts rows from the 3 toy locus regions using tabix
- Writes to tests/toy_3locus/data/raw/
